\sectionframe{Cierre Clase 24}{Transfer learning + el proyecto Coca vs Pepsi}

\begin{frame}[shrink]{Recap: lo que dejamos a medias}
  \vspace{6pt}
  {\footnotesize\color{arcaDark}\bfseries
    En clase 24 vimos que entrenar una CNN desde cero con 4\,000 fotos
    no alcanza para resolver problemas reales. La soluci\'on:
    \textbf{transfer learning}.}

  \vspace{14pt}
  {\footnotesize
  \begin{itemize}\setlength{\itemsep}{6pt}
    \item[{\color{arcaRed}\scriptsize\faArrowRight}]
      \textbf{Backbone preentrenado} (VGG16 en ImageNet 1.4M
      im\'agenes) tiene filtros que ya saben detectar bordes,
      texturas, formas.
    \item[{\color{arcaRed}\scriptsize\faArrowRight}]
      \textbf{Congelamos el backbone} y solo entrenamos un cabezal
      Dense con nuestras pocas fotos.
    \item[{\color{codeGreen}\scriptsize\faArrowRight}]
      Resultado: con 50 fotos por clase logramos $\sim$90\% en Coca vs Pepsi.
    \item[{\color{codeGreen}\scriptsize\faArrowRight}]
      Hoy extendemos esa idea a \textbf{detecci\'on}: misma idea,
      cambia la arquitectura.
  \end{itemize}
  }
\end{frame}

\begin{frame}[shrink]{Transfer Learning: La Idea Visual}
  \vspace{4pt}
  \begin{center}
    \includegraphics[width=0.95\textwidth]{fig_finetune.png}
  \end{center}

  \vspace{4pt}
  {\scriptsize\color{textMuted}
    \faLightbulb\hspace{4pt} Una CNN preentrenada \textbf{sabe ver}:
    bordes, texturas, formas. Reusamos esos filtros (congelados) y
    entrenamos un cabezal nuevo con \emph{nuestras} clases.}
\end{frame}

\begin{frame}[shrink]{?`Por Qu\'e Funciona?}
  \vspace{8pt}
  {\footnotesize\color{arcaDark}\bfseries
    Las primeras capas convolucionales aprenden conceptos que son
    \textbf{los mismos en cualquier tarea de visi\'on}:}

  \vspace{14pt}
  {\footnotesize
  \begin{itemize}\setlength{\itemsep}{6pt}
    \item[{\color{arcaRed}\scriptsize\faArrowRight}]
      \textbf{Capas tempranas:} bordes, manchas, gradientes simples.
    \item[{\color{arcaRed}\scriptsize\faArrowRight}]
      \textbf{Capas medias:} texturas, esquinas, patrones repetitivos.
    \item[{\color{arcaRed}\scriptsize\faArrowRight}]
      \textbf{Capas tard\'ias:} partes (orejas, ojos, ruedas) y luego objetos.
    \item[{\color{codeGreen}\scriptsize\faArrowRight}]
      Solo las \emph{\'ultimas} capas son espec\'ificas de la tarea.
      Reusamos las primeras y reentrenamos las \'ultimas.
  \end{itemize}
  }
\end{frame}

\begin{frame}[fragile,shrink]{C\'odigo Coca vs Pepsi (clase 24)}
  \vspace{-4pt}
  \begin{pyblock}
from tensorflow.keras.applications import VGG16
from tensorflow.keras import Sequential, layers

# 1. Cargar VGG16 sin la cabeza, con pesos de ImageNet
base = VGG16(input_shape=(96,96,3), include_top=False, weights="imagenet")
base.trainable = False                  # congelar capas conv

# 2. Backbone congelado + cabeza nueva
modelo = Sequential([
    base,
    layers.Flatten(),
    layers.Dense(64, activation="relu"),
    layers.Dense(2, activation="softmax"),    # Coca vs Pepsi
])

# 3. Solo entrena el cabezal
modelo.compile("adam", "sparse_categorical_crossentropy", ["accuracy"])
modelo.fit(X_train, y_train, epochs=5, validation_split=0.1)
  \end{pyblock}
\end{frame}

\begin{frame}[shrink]{Limitaci\'on de Esa Aproximaci\'on}
  \vspace{6pt}
  {\footnotesize\color{arcaDark}\bfseries
    El modelo de Coca vs Pepsi es de \textbf{clasificaci\'on}: dice
    ``esta foto es Coca'' o ``es Pepsi''. \textbf{Una respuesta para
    toda la imagen}.}

  \vspace{14pt}
  {\footnotesize
  \begin{itemize}\setlength{\itemsep}{6pt}
    \item[{\color{arcaRed}\scriptsize\faArrowRight}]
      ?`Y si tengo \textbf{varias} latas en una foto?
      No las puede contar.
    \item[{\color{arcaRed}\scriptsize\faArrowRight}]
      ?`Y si quiero saber \textbf{d\'onde} est\'a cada una?
      No las puede ubicar.
    \item[{\color{arcaRed}\scriptsize\faArrowRight}]
      ?`Y si necesito \textbf{leer} la fecha de vencimiento o un texto
      en la lata? No puede tampoco.
    \item[{\color{codeGreen}\scriptsize\faArrowRight}]
      Para todo eso hace falta otra tarea: \textbf{detecci\'on de objetos}.
  \end{itemize}
  }
\end{frame}

\begin{frame}[shrink]{IoU: ?`Qu\'e Tan Buena Es Una Predicci\'on?}
  \vspace{4pt}
  {\footnotesize\color{arcaDark}\bfseries
    \textbf{IoU} (Intersection over Union) compara la caja real con la
    predicha: $\text{IoU} = \dfrac{\text{intersecci\'on}}{\text{uni\'on}}$. Va de 0 a 1.}

  \vspace{6pt}
  \begin{center}
    \includegraphics[width=0.95\textwidth]{fig_iou.png}
  \end{center}

  \vspace{2pt}
  {\scriptsize\color{textMuted}
    \faLightbulb\hspace{4pt} Convenci\'on: IoU $\geq$ 0.5 cuenta como
    detecci\'on correcta. Es el umbral de la m\'etrica m\'as usada,
    \textbf{mAP@0.5}.}
\end{frame}

\begin{frame}[shrink]{?`Por Qu\'e una CNN Cl\'asica No Sirve?}
  \vspace{8pt}
  {\footnotesize\color{arcaDark}\bfseries
    Una CNN cl\'asica produce \textbf{una sola salida} por imagen.
    En detecci\'on tenemos que predecir \textbf{N salidas variables} ---
    una por objeto detectado.}

  \vspace{14pt}
  {\footnotesize
  \begin{tabular}{@{}p{4.0cm} p{4.0cm} p{4.0cm}@{}}
    \toprule
    \textbf{Problema} & \textbf{CNN cl\'asica} & \textbf{Detector} \\
    \midrule
    \rowcolor{arcaGray}
    Salida
    & 1 vector fijo
    & Lista variable de bboxes \\
    Cantidad de objetos
    & Asume 1
    & Puede ser 0, 1 o 100 \\
    \rowcolor{arcaGray}
    Ubicaci\'on
    & No la sabe
    & Da (x, y, w, h) por objeto \\
    \bottomrule
  \end{tabular}
  }

  \vspace{14pt}
  \begin{tcolorbox}[colback=arcaRed!8, colframe=arcaRed, boxrule=0.5pt,
    arc=2pt, left=6pt, right=6pt, top=4pt, bottom=4pt]
    {\scriptsize\color{arcaDark}
    \faExclamationTriangle\hspace{4pt} Hace falta una arquitectura distinta,
    dise\~nada para producir m\'ultiples bboxes en una sola pasada.
    Esa es la idea de \textbf{YOLO}.}
  \end{tcolorbox}
\end{frame}

%  BLOQUE D: YOLO EN PROFUNDIDAD
% =============================================================================
\sectionframe{C\'omo Funciona YOLO}{La idea revolucionaria de 2015}

\begin{frame}[shrink]{La Idea: ``You Only Look Once''}
  \vspace{6pt}
  {\footnotesize\color{arcaDark}\bfseries
    Antes de YOLO los detectores hac\'ian dos pasadas: primero proponer
    regiones, despu\'es clasificar cada una. \textbf{Lento}.
    YOLO lo hace todo en \textbf{una sola pasada} de la CNN.}

  \vspace{14pt}
  {\footnotesize
  \begin{enumerate}\setlength{\itemsep}{6pt}
    \item[{\color{arcaRed}\bfseries 1.}]
      \textbf{Divide la imagen en una rejilla} (ej: 13$\times$13).
    \item[{\color{arcaRed}\bfseries 2.}]
      Cada celda predice: ``?`hay un objeto centrado en m\'i?'' Si s\'i,
      devuelve una bbox completa (x, y, w, h, clase, confianza).
    \item[{\color{arcaRed}\bfseries 3.}]
      \textbf{Una sola pasada} de la red $\Rightarrow$ todas las cajas a la vez.
    \item[{\color{codeGreen}\bfseries 4.}]
      Al final, \textbf{NMS} limpia las cajas duplicadas.
  \end{enumerate}
  }
\end{frame}

\begin{frame}[shrink]{La Rejilla en la Pr\'actica}
  \vspace{4pt}
  \begin{center}
    \includegraphics[width=0.85\textwidth]{fig_yolo_grid.png}
  \end{center}

  \vspace{4pt}
  {\scriptsize\color{textMuted}
    \faLightbulb\hspace{4pt} La celda donde cae el centro de la placa
    es \emph{responsable} de detectarla.
    En YOLO26 hay 3 escalas de rejilla simult\'aneas
    para detectar objetos de distintos tama\~nos.}
\end{frame}

\begin{frame}[shrink]{NMS: Limpiar Detecciones Duplicadas}
  \vspace{4pt}
  {\footnotesize\color{arcaDark}\bfseries
    YOLO produce \textbf{cientos} de detecciones por imagen, varias
    del mismo objeto. \textbf{NMS} (Non-Maximum Suppression) deja una sola.}

  \vspace{6pt}
  \begin{center}
    \includegraphics[width=0.95\textwidth]{fig_nms.png}
  \end{center}

  \vspace{2pt}
  {\scriptsize\color{textMuted}
    \faLightbulb\hspace{4pt} Algoritmo: ordenar por confianza, quedarse
    con la mejor, descartar las que solapan con IoU $>$ 0.5. Repetir.}
\end{frame}

\sectionframe{M\'etricas}{C\'omo medir si el modelo es bueno}

\begin{frame}[shrink]{La M\'etrica Reina: mAP}
  \vspace{8pt}
  {\footnotesize\color{arcaDark}\bfseries
    \textbf{mAP} (mean Average Precision) es el ``accuracy'' de detecci\'on.
    Combina precision y recall sobre todas las clases.}

  \vspace{14pt}
  {\footnotesize
  \begin{itemize}\setlength{\itemsep}{6pt}
    \item[{\color{arcaRed}\scriptsize\faArrowRight}]
      \textbf{mAP@0.5}: una detecci\'on cuenta como correcta si su IoU
      con la real es $\geq$ 0.5. \textbf{La m\'as usada}.
    \item[{\color{arcaRed}\scriptsize\faArrowRight}]
      \textbf{mAP@0.5:0.95}: promedio sobre 10 umbrales (0.5 a 0.95).
      M\'as estricta. La que reporta COCO.
    \item[{\color{codeGreen}\scriptsize\faArrowRight}]
      \textbf{Para nuestro proyecto:} mAP@0.5 $\geq$ 0.85 con $\sim$300 fotos
      etiquetadas + fine-tune.
  \end{itemize}
  }
\end{frame}

\begin{frame}[shrink]{Precision, Recall y Trade-offs}
  \vspace{8pt}
  {\footnotesize
  \begin{tabular}{@{}p{2.8cm} p{4.0cm} p{5.5cm}@{}}
    \toprule
    \textbf{M\'etrica} & \textbf{F\'ormula} & \textbf{Pregunta que responde} \\
    \midrule
    \rowcolor{arcaGray}
    \textbf{Precision} & TP / (TP + FP)
    & De las cajas que detect\'e, ?`cu\'antas eran reales? \\
    \textbf{Recall} & TP / (TP + FN)
    & De las cajas reales, ?`cu\'antas detect\'e? \\
    \rowcolor{arcaGray}
    \textbf{F1}        & 2 PR / (P + R)
    & Equilibrio entre las dos. \\
    \bottomrule
  \end{tabular}
  }

  \vspace{14pt}
  \begin{tcolorbox}[colback=arcaGray, colframe=arcaRed, boxrule=0.5pt,
    arc=2pt, left=6pt, right=6pt, top=4pt, bottom=4pt]
    {\scriptsize\color{arcaDark}
    \faKey\hspace{4pt}\textbf{Trade-off cl\'asico}: bajar el umbral de
    confianza (\texttt{conf}) sube recall (m\'as cajas), pero baja precision
    (m\'as falsos positivos). Hay que elegir seg\'un el caso.}
  \end{tcolorbox}
\end{frame}

\begin{frame}[shrink]{Pretrained YOLO26: ?`Detecta Placas?}

  \vspace{14pt}

  \begin{itemize}\setlength{\itemsep}{6pt}
    \item[{\color{arcaRed}\scriptsize\faArrowRight}]
      Pretrained detecta el \texttt{auto} pero \textbf{no la placa}.
    \item[{\color{arcaRed}\scriptsize\faArrowRight}]
      Despu\'es de fine-tune con $\sim$300 fotos etiquetadas $\to$ mAP@0.5 $\geq$ 0.85.
    \item[{\color{codeGreen}\scriptsize\faArrowRight}]
      \textbf{Demo dramatica}: pretrained = 0\% en placas $\to$ fine-tune = 90\%+.
  \end{itemize}
  }

\begin{frame}[fragile,shrink]{Fine-Tune YOLO26 en 4 L\'ineas}
  \vspace{-4pt}
  \begin{pyblock}
from ultralytics import YOLO

# 1. Cargar pretrained (YOLO26 nano)
model = YOLO("yolo26n.pt")

# 2. Fine-tune con NUESTROS datos
model.train(data="dataset/data.yaml",
            epochs=30, imgsz=640, batch=16)

# 3. Evaluar
metrics = model.val()
print(f"mAP@0.5 = {metrics.box.map50:.3f}")
  \end{pyblock}

  \vspace{6pt}
  {\scriptsize\color{textMuted}
    \faLightbulb\hspace{4pt} Estructura esperada:
    \texttt{dataset/images/\{train,val\}/*.jpg} +
    \texttt{dataset/labels/\{train,val\}/*.txt} + un \texttt{data.yaml}.
    CVAT lo exporta listo en formato YOLO 1.1.}
\end{frame}

\begin{frame}[fragile,shrink]{Despu\'es de YOLO: OCR}
  \vspace{-4pt}
  \begin{pyblock}
import easyocr

# 1. Detectar placa con YOLO fine-tuneado
results = model("auto.jpg")
boxes = results[0].boxes.xyxy   # cajas (x1,y1,x2,y2)

# 2. Recortar la región de la placa
img = cv2.imread("auto.jpg")
x1, y1, x2, y2 = map(int, boxes[0])
plate_crop = img[y1:y2, x1:x2]

# 3. Pasar el recorte por OCR
reader = easyocr.Reader(['en', 'es'])
texto = reader.readtext(plate_crop, detail=0)
print(texto)    # ['PCJ-3421']
  \end{pyblock}

  \vspace{4pt}
  {\scriptsize\color{textMuted}
    \faLightbulb\hspace{4pt} \textbf{Pipeline en cascada:} 2 modelos
    independientes encadenados. Si uno falla, el otro tampoco funciona ---
    pero cada uno es bueno en lo suyo.}
\end{frame}
