% F12 — PDF escaneado: PyPDF devuelve vacío
\begin{frame}[fragile, shrink]{PDF escaneado --- PyPDF devuelve nada}
  \begin{pyblock}

# 0 caracteres extraidos
  \end{pyblock}
  \vspace{6pt}
  \begin{tcolorbox}[colback=arcaRed!8, colframe=arcaRed, leftrule=3pt,
    boxrule=0pt, sharp corners, left=8pt, right=8pt, top=4pt, bottom=4pt]
    {\footnotesize\color{arcaDark}\bfseries
    Cero caracteres. PyPDF no inventa texto, lee lo que está en el PDF. En el
    escaneado el contenido vive como \emph{imagen} dentro de cada página.}
  \end{tcolorbox}
  \vspace{6pt}
  {\footnotesize\color{arcaDark}
  Necesitamos un paso previo: convertir cada página a una imagen (\texttt{pymupdf})
  y pedirle a un \textbf{modelo de visión (VLM)} que ``lea'' la imagen y devuelva
  texto. Eso es OCR moderno.}
\end{frame}

% F13 — Convertir página a imagen con pymupdf
\begin{frame}[fragile, shrink]{Paso intermedio --- página $\to$ imagen con \texttt{pymupdf}}
  \begin{pyblock}
import fitz, base64
doc = fitz.open("manual_ESCANEADO.pdf")
pix = doc[0].get_pixmap(dpi=140)        # renderiza pagina 1
png_bytes = pix.tobytes("png")
b64 = base64.b64encode(png_bytes).decode()
print(f"PNG: {len(png_bytes)} bytes,  base64: {len(b64)} chars")
  \end{pyblock}
  \vspace{6pt}
  \begin{itemize}\setlength{\itemsep}{3pt}
    \footnotesize
    \item \texttt{pymupdf} (módulo \texttt{fitz}): rendea cada página como mapa de pixels.
    \item \texttt{dpi=140} es buen balance calidad/tamaño para OCR.
    \item \textbf{Base64} convierte los bytes binarios a string para mandar al API.
  \end{itemize}
\end{frame}
